## <T k="doc.usage" />

```tsx
import { OverflowList, Tag, Button, Popover, PopoverTrigger, PopoverContent, Stack, VisuallyHidden } from "wimui";

const Amenities = ({ amenities }: { amenities: string[] }) => (
  <OverflowList
    role="group"
    aria-label="Amenities"
    items={amenities}
    renderItem={(amenity) => (
      <Tag intent="neutral" variant="subtle" size="sm">
        {amenity}
      </Tag>
    )}
    renderOverflow={(hidden) => (
      <Popover placement="bottom-start">
        <PopoverTrigger asChild>
          <Button size="sm" variant="ghost">
            +{hidden.length}
            <VisuallyHidden> Show all amenities</VisuallyHidden>
          </Button>
        </PopoverTrigger>
        <PopoverContent>
          <Stack gap="xs" align="start">
            {hidden.map((amenity) => (
              <Tag key={amenity} intent="neutral" variant="subtle" size="sm">
                {amenity}
              </Tag>
            ))}
          </Stack>
        </PopoverContent>
      </Popover>
    )}
  />
);
```
