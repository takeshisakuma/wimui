## <T k="doc.appshell_mobile_drawer_title" />

<p><T k="doc.appshell_mobile_drawer_desc" /></p>

```tsx
function AppLayout() {
  const [mobileOpen, setMobileOpen] = useState(false);

  return (
    <AppShell
      header={
        <Header bordered>
          <Header.Section align="start">
            {/* Visible only at or below the md breakpoint */}
            <HamburgerMenu
              visibleBelow="md"
              open={mobileOpen}
              onClick={() => setMobileOpen((o) => !o)}
            />
          </Header.Section>
        </Header>
      }
      sidebar={
        <Sidebar
          mobileOpen={mobileOpen}
          onOverlayClick={() => setMobileOpen(false)}
        >
          <Sidebar.Content>{/* Sidebar items */}</Sidebar.Content>
        </Sidebar>
      }
    >
      {/* Main content */}
    </AppShell>
  );
}
```
