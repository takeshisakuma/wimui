## <T k="doc.usage" />

<p><T k="doc.confirm_usage_desc" /></p>

```tsx
import { Button, ConfirmProvider, useConfirm, WimProvider } from "wimui";

// Once, near the root
<WimProvider>
  <ConfirmProvider>
    <App />
  </ConfirmProvider>
</WimProvider>;

// Anywhere below it
const DeleteButton = ({ id }: { id: string }) => {
  const confirm = useConfirm();
  const onDelete = async () => {
    const ok = await confirm({
      title: "Delete “Q3-report.pdf”?",
      description: "The 6 people it is shared with lose access. This cannot be undone.",
      confirmLabel: "Delete",
      intent: "danger",
    });
    if (ok) await deleteFile(id);
  };
  return <Button variant="ghost" intent="danger" onClick={onDelete}>Delete</Button>;
};
```
